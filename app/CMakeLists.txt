cmake_minimum_required(VERSION 3.20.0)

# Matter (connectedhomeip) is fetched by west into modules/lib/connectedhomeip.
# Its Zephyr module lives in a sub-directory, so it has to be registered
# explicitly before Zephyr is loaded.
if(NOT DEFINED CHIP_ROOT)
  get_filename_component(CHIP_ROOT $ENV{ZEPHYR_BASE}/../modules/lib/connectedhomeip ABSOLUTE)
endif()
list(APPEND EXTRA_ZEPHYR_MODULES ${CHIP_ROOT}/config/zephyr/chip-module)

# Normally generated by the Matter bootstrap (scripts/activate.sh). It is only
# needed for pigweed's clang toolchain, so an empty file is enough here.
if(NOT EXISTS ${CHIP_ROOT}/build_overrides/pigweed_environment.gni)
  file(WRITE ${CHIP_ROOT}/build_overrides/pigweed_environment.gni "")
endif()

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})

project(app LANGUAGES C CXX)

# libCHIP.a uses kernel APIs (e.g. k_timer) and is placed after the kernel
# library on the link line; append the kernel archive again after it.
target_link_libraries(chip INTERFACE $<TARGET_FILE:kernel>)

target_sources(app PRIVATE
  src/main.cpp
  src/light.cpp
)
target_sources_ifdef(CONFIG_WIFI_CREDENTIALS_CONNECT_STORED app PRIVATE src/wifi.cpp)
target_include_directories(app PRIVATE src)

include(${CHIP_ROOT}/src/app/chip_data_model.cmake)

chip_configure_data_model(app
  ZAP_FILE ${CMAKE_CURRENT_SOURCE_DIR}/light.zap
  ZCL_PATH ${CHIP_ROOT}/src/app/zap-templates/zcl/zcl.json
)

# The code below locates the git index file for this repository and adds it as a dependency for
# the application VERSION file so that if the repo has a new commit added, even if no files in
# the build have changed, the application version file will be regenerated with the new git commit
find_package(Git QUIET)
if(GIT_FOUND)
  execute_process(
    COMMAND ${GIT_EXECUTABLE} rev-parse --absolute-git-dir
    WORKING_DIRECTORY .
    OUTPUT_VARIABLE application_git_dir
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_STRIP_TRAILING_WHITESPACE
    ERROR_VARIABLE stderr
    RESULT_VARIABLE return_code
  )
  # If there is an error e.g. it is not a git repo, we just silently ignore it and continue
  # without a dependency, this will be the case with freestanding applications.
  if(NOT return_code)
    if(NOT "${stderr}" STREQUAL "")
      message(WARNING "Application build version git rev-parse warned: ${stderr}")
    endif()

    set_property(TARGET app_version_h PROPERTY APP_VERSION_DEPENDS ${application_git_dir}/index)
  endif()
endif()
